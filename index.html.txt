  <script>
    const API_KEY = '1f672161192366998d6d0f1591b34086';
    const TRENDING_URL = `https://api.themoviedb.org/3/trending/movie/week?api_key=${API_KEY}`;
    const IMG_URL = 'https://image.tmdb.org/t/p/w500';
    const ORIGINAL_IMG_URL = 'https://image.tmdb.org/t/p/original';

    let heroMovies = [];
    let currentHeroIndex = 0;
    let heroInterval;

    async function loadApp() {
      try {
        const res = await fetch(TRENDING_URL);
        const data = await res.json();
        heroMovies = data.results;

        if (heroMovies && heroMovies.length > 0) {
          updateHeroBanner();
          if (heroInterval) clearInterval(heroInterval);
          heroInterval = setInterval(slideHeroBanner, 4000);

          showMovies(heroMovies);
        }
      } catch (err) {
        console.error('Error loading movies:', err);
        document.getElementById('movies').innerHTML = '<p class="col-span-full text-center text-red-500">Failed to load movies. Please check your connection.</p>';
      }
    }

    function updateHeroBanner() {
      if(!heroMovies.length) return;
      const topMovie = heroMovies[currentHeroIndex];
      document.getElementById('heroBanner').style.backgroundImage = `url('${ORIGINAL_IMG_URL + topMovie.backdrop_path}')`;
      document.getElementById('heroTitle').innerText = topMovie.title;
      document.getElementById('heroDesc').innerText = topMovie.overview;
      // Using VidLink Pro stable embed link
      document.getElementById('heroPlayBtn').onclick = () => window.open(`https://vidlink.pro/movie/${topMovie.id}`, '_blank');
    }

    function slideHeroBanner() {
      currentHeroIndex = (currentHeroIndex + 1) % Math.min(heroMovies.length, 5);
      updateHeroBanner();
    }

    function showMovies(movies) {
      const container = document.getElementById('movies');
      container.innerHTML = '';
      movies.forEach(movie => {
        const card = document.createElement('div');
        card.className = 'movie-card bg-zinc-900 border border-zinc-800/80 rounded-2xl overflow-hidden cursor-pointer shadow-lg';
        card.innerHTML = `
          <img src="${IMG_URL + movie.poster_path}" alt="${movie.title}" class="w-full h-56 sm:h-64 object-cover">
          <div class="p-3">
            <h4 class="font-bold text-sm truncate text-zinc-100">${movie.title}</h4>
            <div class="flex justify-between items-center mt-1.5">
              <span class="text-xs text-amber-400 font-semibold">⭐ ${movie.vote_average ? movie.vote_average.toFixed(1) : '0.0'}</span>
              <span class="text-[10px] bg-zinc-800 text-zinc-300 px-2 py-0.5 rounded-md font-medium">HD</span>
            </div>
          </div>
        `;
        // Using VidLink Pro stable embed link for cards
        card.onclick = () => window.open(`https://vidlink.pro/movie/${movie.id}`, '_blank');
        container.appendChild(card);
      });
    }

    function toggleSearch() {
      const modal = document.getElementById('searchModal');
      const isHidden = modal.classList.contains('hidden');
      if(isHidden) {
        modal.classList.remove('hidden');
        modal.classList.add('flex');
        document.getElementById('searchInput').focus();
      } else {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
      }
    }

    function toggleMenu() {
      const menu = document.getElementById('sideMenu');
      const overlay = document.getElementById('sideMenuOverlay');
      const isClosed = menu.classList.contains('translate-x-full');
      
      if(isClosed) {
        menu.classList.remove('translate-x-full');
        overlay.classList.remove('hidden');
      } else {
        menu.classList.add('translate-x-full');
        overlay.classList.add('hidden');
      }
    }

    function toggleAuthModal() {
      const modal = document.getElementById('authModal');
      modal.classList.toggle('hidden');
      modal.classList.toggle('flex');
    }

    function handleSignIn() {
      const email = document.getElementById('userEmail').value;
      if(!email) {
        alert('Please enter your email or username');
        return;
      }
      alert(`Welcome back, ${email}! Signed in successfully to DN MOVIES.`);
      toggleAuthModal();
    }

    async function searchMovies(query) {
      if(!query) {
        document.getElementById('searchResults').innerHTML = '';
        return;
      }
      try {
        const res = await fetch(`https://api.themoviedb.org/3/search/movie?api_key=${API_KEY}&query=${query}`);
        const data = await res.json();
        const container = document.getElementById('searchResults');
        container.innerHTML = '';
        
        if(!data.results || data.results.length === 0) {
          container.innerHTML = '<p class="col-span-full text-center text-zinc-500 text-sm">No movies found.</p>';
          return;
        }

        data.results.forEach(movie => {
          if(!movie.poster_path) return;
          const card = document.createElement('div');
          card.className = 'movie-card bg-zinc-900 border border-zinc-800 rounded-2xl overflow-hidden cursor-pointer flex gap-3 p-2.5 items-center';
          card.innerHTML = `
            <img src="${IMG_URL + movie.poster_path}" class="w-16 h-20 object-cover rounded-xl shrink-0">
            <div class="overflow-hidden">
              <h5 class="font-bold text-sm text-white truncate">${movie.title}</h5>
              <p class="text-xs text-amber-400 mt-1">⭐ ${movie.vote_average ? movie.vote_average.toFixed(1) : 'N/A'}</p>
            </div>
          `;
          card.onclick = () => window.open(`https://vidlink.pro/movie/${movie.id}`, '_blank');
          container.appendChild(card);
        });
      } catch (err) {
        console.error('Search error:', err);
      }
    }

    loadApp();
  </script>
